#include <stdio.h>

int main()
{
    int w;
    scanf("%d", &w);
    int a = 0;
    int j;
    for (int i = 1; i < w; i++)
    {
        j = w - i;
        if (i % 2 == 0 && j % 2 == 0)
        {
            a++;
        }
    }
    if (a > 0)
    {
        printf("YES");
    }
    else
    {
        printf("NO");
    }
    return 0;
}
